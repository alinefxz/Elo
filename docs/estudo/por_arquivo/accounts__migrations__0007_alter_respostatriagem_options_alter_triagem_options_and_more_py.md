def preencher_dados_da_triagem(apps, schema_editor):
    """Converte os registros da versão inicial para os campos novos."""

    Triagem = apps.get_model("accounts", "Triagem")
    RespostaTriagem = apps.get_model("accounts", "RespostaTriagem")

    # Uma triagem antiga com data final já estava concluída.
    Triagem.objects.filter(
        finalizada_em__isnull=False,
    ).update(status="CONCLUIDA")

    for resposta in RespostaTriagem.objects.all().iterator():
